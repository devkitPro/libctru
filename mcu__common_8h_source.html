273static inline u8 mcuTicksFromMs(u32 ms)
274{
275 // 512Hz
276 u32 res = 512u * ms / 1000u;
277 res = res < 2 ? 1 : res - 1; // res not allowed to be zero
278 res = res > 255 ? 255 : res; // res can't exceed 255
279 return (u8)res;
280}
